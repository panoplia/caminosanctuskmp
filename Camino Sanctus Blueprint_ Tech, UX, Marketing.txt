You would end up writing all of the critical business logic in native code anyway and maintaining a Flutter shell for UI that adds 40MB+ to your binary for no reason.

### Android: Accessibility Service Architecture

This is the core of how you intercept app launches on Android. There is no other reliable mechanism.

// CaminoAccessibilityService.kt
class CaminoAccessibilityService : AccessibilityService() {

   private lateinit var interceptEngine: InterceptEngine
   private lateinit var overlayManager: InterceptOverlayManager
   private val monitoredPackages = mutableSetOf<String>()
   
   // Debounce to prevent double-fire on app transitions
   private var lastInterceptTimestamp = 0L
   private val DEBOUNCE_MS = 3000L
   
   // Cooldown per app to prevent rage-loop frustration
   private val appCooldowns = mutableMapOf<String, Long>()
   private val COOLDOWN_MS = 1800_000L // 30 minutes

   override fun onServiceConnected() {
       val info = AccessibilityServiceInfo().apply {
           eventTypes = AccessibilityEvent.TYPE_WINDOW_STATE_CHANGED
           feedbackType = AccessibilityServiceInfo.FEEDBACK_GENERIC
           flags = AccessibilityServiceInfo.FLAG_INCLUDE_NOT_IMPORTANT_VIEWS
           notificationTimeout = 100
       }
       serviceInfo = info
       
       // Load user's selected apps from local DB
       monitoredPackages.addAll(
           interceptEngine.getMonitoredPackages()
       )
   }

   override fun onAccessibilityEvent(event: AccessibilityEvent?) {
       if (event?.eventType != AccessibilityEvent.TYPE_WINDOW_STATE_CHANGED) return
       
       val packageName = event.packageName?.toString() ?: return
       val now = System.currentTimeMillis()
       
       // Guard clauses
       if (packageName !in monitoredPackages) return
       if (now - lastInterceptTimestamp < DEBOUNCE_MS) return
       if (appCooldowns[packageName]?.let { now - it < COOLDOWN_MS } == true) return
       
       lastInterceptTimestamp = now
       
       // Core intercept logic
       val userState = interceptEngine.getCurrentState()
       val contentCard = interceptEngine.selectCard(
           context = InterceptContext(
               targetApp = packageName,
               timeOfDay = Calendar.getInstance().get(Calendar.HOUR_OF_DAY),
               currentStreak = userState.streak,
               depthTier = userState.depthTier
           )
       )
       
       // Launch overlay
       overlayManager.showIntercept(
           card = contentCard,
           onComplete = { duration, wasSkipped ->
               interceptEngine.logEvent(
                   InterceptEvent(
                       targetApp = packageName,
                       durationSeconds = duration,
                       completed = !wasSkipped,
                       contentCardId = contentCard.id,
                       timestamp = now
                   )
               )
               if (!wasSkipped) {
                   appCooldowns[packageName] = now
               }
           }
       )
   }
}

// DeviceActivityMonitorExtension.swift
import DeviceActivity
import ManagedSettings

class CaminoDeviceActivityMonitor: DeviceActivityMonitor {
   
   let store = ManagedSettingsStore()
   
   override func intervalDidStart(for activity: DeviceActivityName) {
       // Called when a monitored time window begins
       // Shield all selected applications
       let model = CaminoModel.shared
       store.shield.applications = model.selectedApps
       store.shield.applicationCategories = 
           ShieldSettings.ActivityCategoryPolicy.specific(model.selectedCategories)
   }
   
   override func intervalDidEnd(for activity: DeviceActivityName) {
       // Monitoring window ended — remove all shields
       store.shield.applications = nil
       store.shield.applicationCategories = nil
   }
   
   override func eventDidReachThreshold(
       _ event: DeviceActivityEvent.Name, 
       activity: DeviceActivityName
   ) {
       // Usage threshold reached — re-engage shield
       store.shield.applications = CaminoModel.shared.selectedApps
   }
}

// Shared App Group data pipeline
let sharedDefaults = UserDefaults(suiteName: "group.com.panoplia.caminosanctus")

// Main app writes:
func preloadShieldContent() {
   let nextCards = contentEngine.preSelectCards(count: 20)
   let encoded = try? JSONEncoder().encode(nextCards)
   sharedDefaults?.set(encoded, forKey: "shield_content_queue")
   sharedDefaults?.set(0, forKey: "shield_content_index")
}

// Extension reads:
func getNextCard() -> ContentCard? {
   guard let data = sharedDefaults?.data(forKey: "shield_content_queue"),
         var index = sharedDefaults?.integer(forKey: "shield_content_index"),
         let cards = try? JSONDecoder().decode([ContentCard].self, from: data) else {
       return ContentCard.fallback // Always have a hardcoded fallback
   }
   let card = cards[index % cards.count]
   index += 1
   sharedDefaults?.set(index, forKey: "shield_content_index")
   return card
}

// Using llama.cpp Android bindings via llamakt
class LocalSLMEngine(private val context: Context) {
   
   private var model: LlamaModel? = null
   
   suspend fun initialize() = withContext(Dispatchers.IO) {
       val modelPath = context.filesDir.resolve("tinyllama-theology-q4km.gguf")
       
       if (!modelPath.exists()) {
           // Download on first premium activation (~670MB)
           // Show progress UI during download
           downloadModel(modelPath)
       }
       
       model = LlamaModel(
           modelPath = modelPath.absolutePath,
           contextSize = 2048,
           threads = 4,       // Use performance cores
           gpuLayers = 0      // CPU-only for compatibility
       )
   }
   
   suspend fun generate(
       systemPrompt: String,
       conversationHistory: List<Message>,
       maxTokens: Int = 400
   ): String = withContext(Dispatchers.IO) {
       
       val prompt = buildPrompt(systemPrompt, conversationHistory)
       
       model?.generate(
           prompt = prompt,
           maxTokens = maxTokens,
           temperature = 0.7f,
           topP = 0.9f,
           stopSequences = listOf("\nUsuario:", "\n[END]")
       ) ?: throw IllegalStateException("Model not loaded")
   }
   
   private fun buildPrompt(
       system: String, 
       history: List<Message>
   ): String = buildString {
       append("<|system|>\n$system</s>\n")
       history.forEach { msg ->
           when (msg.role) {
               Role.USER -> append("<|user|>\n${msg.content}</s>\n")
               Role.ASSISTANT -> append("<|assistant|>\n${msg.content}</s>\n")
           }
       }
       append("<|assistant|>\n")
   }
}

## PILLAR 2: UX MECHANICS & THE CORE LOOP

### The Psychological Analysis You Didn't Ask For But Need

**Prayer Lock's punitive model works because of three psychological mechanisms:**

1. **Implementation Intention Enforcement** (Gollwitzer, 1999): Users form an "if-then" plan ("if I try to open Instagram, then I must pray first"). The app externalizes willpower. This is not punishment — it is pre-commitment device theory. It works the same way Ulysses binding himself to the mast works.
2. **Loss Aversion as Activation Energy** (Kahneman & Tversky): The "cost" of not praying (being locked out of Instagram) is felt more acutely than the "benefit" of praying. This drives initial engagement.
3. **Variable Ratio Reward Absence**: Prayer Lock does NOT reward the prayer. There's no streak, no gamification, no celebration screen. This is accidentally brilliant because it avoids the user psychologically "paying" for Instagram access with a hollow ritual. The prayer is the only event. This keeps the spiritual framing intact.

**However, the punitive model has a fatal flaw for your market:**

The churn curve on punitive apps follows a predictable pattern:

**The LatAm-specific variable**: Latin American Christian culture — especially Catholic culture — has a fundamentally different spiritual psychology than US evangelical culture. US evangelical culture is **transactional** ("I did the prayer, I earned the blessing"). LatAm Catholic culture is **relational and devotional** ("I seek encounter with the divine"). A purely punitive mechanic will feel incongruent to your core audience. They'll use it, but they'll resent it subtly, and churn will be 15-20% higher than Prayer Lock's US numbers.

function shouldRequestReview(user: UserState): boolean {
// ── GATE 1: Maturity ──────────────────────────────
// User must have enough sessions to have formed a habit
// and genuine appreciation. Too early = low rating.
const hasMinSessions = user.completedSessions >= 5;
// ── GATE 2: Recency of Streak ─────────────────────
// User is currently on an active streak (not returning
// after a lapse, when frustration/guilt may be present)
const isOnActiveStreak = user.currentStreak >= 3;
// ── GATE 3: Just Completed a Milestone ────────────
// The completion screen of their 5th, 7th, or 10th
// session — these are the moments of highest
// self-identity reinforcement
const justHitMilestone = [5, 7, 10, 14, 21, 30].includes(
user.completedSessions
);
// ── GATE 4: Positive Interaction Signal ───────────
// User performed a VOLUNTARY positive action in this
// session (shared a verse, saved a reflection, tapped
// the heart/amen button). This is behavioral proof
// of positive sentiment — not assumption.
const gavePositiveSignal = user.thisSession.positiveActions >= 1;
// ── GATE 5: Never Asked Before ────────────────────
// One shot. Make it count.
const neverAskedBefore = !user.reviewPromptShown;
// ── GATE 6: Not Near Paywall ──────────────────────
// NEVER within same session as paywall display.
// Mixing commercial ask + review ask = poison.
const noPaywallThisSession = !user.thisSession.paywallShown;
const noPaywallRecently = daysSince(user.lastPaywallShown) >= 2;
return (
hasMinSessions &&
isOnActiveStreak &&
justHitMilestone &&
gavePositiveSignal &&
neverAskedBefore &&
noPaywallThisSession &&
noPaywallRecently
);
}
